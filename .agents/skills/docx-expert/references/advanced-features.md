# Advanced DOCX Features

## Sections (Different Page Layouts in Same Document)

```python
from docx.enum.section import WD_ORIENT

# First section: portrait
section1 = doc.sections[0]
section1.orientation = WD_ORIENT.PORTRAIT

# Add a new section: landscape (for wide tables/charts)
doc.add_section()
section2 = doc.sections[-1]
section2.orientation = WD_ORIENT.LANDSCAPE
section2.page_width = Inches(11)
section2.page_height = Inches(8.5)

# Back to portrait
doc.add_section()
section3 = doc.sections[-1]
section3.orientation = WD_ORIENT.PORTRAIT
section3.page_width = Inches(8.5)
section3.page_height = Inches(11)
```

## Different Headers/Footers Per Section

```python
section = doc.sections[-1]
section.different_first_page_header_footer = True

# First page header (e.g., cover page — empty)
first_header = section.first_page_header
first_header.paragraphs[0].text = ''

# Regular pages header
header = section.header
header.is_linked_to_previous = False  # Don't inherit from previous section
header.paragraphs[0].text = 'Chapter 2: Analysis'
```

## Cover Page

```python
def add_cover_page(doc, title, subtitle, author, date, logo_path=None):
    """Add a professional cover page."""
    # Add some vertical space
    for _ in range(6):
        doc.add_paragraph()

    if logo_path:
        doc.add_picture(logo_path, width=Inches(2))
        doc.paragraphs[-1].alignment = WD_ALIGN_PARAGRAPH.CENTER
        doc.add_paragraph()

    # Title
    p = doc.add_paragraph()
    p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    run = p.add_run(title)
    run.font.size = Pt(36)
    run.font.bold = True
    run.font.color.rgb = RGBColor(0x1a, 0x36, 0x5d)

    # Subtitle
    p = doc.add_paragraph()
    p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    run = p.add_run(subtitle)
    run.font.size = Pt(16)
    run.font.color.rgb = RGBColor(0x64, 0x74, 0x8b)

    # Spacer
    for _ in range(4):
        doc.add_paragraph()

    # Author and date
    p = doc.add_paragraph()
    p.alignment = WD_ALIGN_PARAGRAPH.CENTER
    run = p.add_run(f'{author}\n{date}')
    run.font.size = Pt(12)
    run.font.color.rgb = RGBColor(0x64, 0x74, 0x8b)

    doc.add_page_break()
```

## Footnotes

python-docx doesn't natively support footnotes. Use XML:

```python
def add_footnote(paragraph, text):
    """Add a footnote to the end of a paragraph."""
    run = paragraph.add_run()
    footnote_id = str(len(paragraph.part.element.findall(qn('w:footnote'))) + 1)

    rPr = run._r.makeelement(qn('w:rPr'), {})
    rStyle = rPr.makeelement(qn('w:rStyle'), {qn('w:val'): 'FootnoteReference'})
    rPr.append(rStyle)
    run._r.append(rPr)

    footnoteRef = run._r.makeelement(qn('w:footnoteReference'), {qn('w:id'): footnote_id})
    run._r.append(footnoteRef)
```

## Bookmarks and Cross-References

```python
def add_bookmark(paragraph, bookmark_name):
    """Add a bookmark at the current paragraph."""
    tag = paragraph._p
    start = tag.makeelement(qn('w:bookmarkStart'), {
        qn('w:id'): '0',
        qn('w:name'): bookmark_name,
    })
    end = tag.makeelement(qn('w:bookmarkEnd'), {qn('w:id'): '0'})
    tag.append(start)
    tag.append(end)
```

## Tabs and Tab Stops

```python
from docx.shared import Inches
from docx.enum.text import WD_TAB_ALIGNMENT, WD_TAB_LEADER

pf = paragraph.paragraph_format
tab_stop = pf.tab_stops.add_tab_stop(Inches(6), WD_TAB_ALIGNMENT.RIGHT, WD_TAB_LEADER.DOTS)

# Use tab in text
paragraph.add_run('Item 1\tPage 5')  # \t inserts a tab
```

## Find and Replace in Existing Documents

```python
def find_and_replace(doc, find_text, replace_text):
    """Replace all occurrences of find_text with replace_text."""
    for paragraph in doc.paragraphs:
        if find_text in paragraph.text:
            for run in paragraph.runs:
                if find_text in run.text:
                    run.text = run.text.replace(find_text, replace_text)
    for table in doc.tables:
        for row in table.rows:
            for cell in row.cells:
                for paragraph in cell.paragraphs:
                    if find_text in paragraph.text:
                        for run in paragraph.runs:
                            if find_text in run.text:
                                run.text = run.text.replace(find_text, replace_text)
```

## Template-Based Generation

**Recommended: use `docxtpl` for template filling** (see styles-and-formatting.md for full docxtpl reference).

```python
# docxtpl approach — preserves all Word formatting exactly
from docxtpl import DocxTemplate
tpl = DocxTemplate('/path/to/template.docx')
tpl.render({'company_name': 'Acme Corp', 'date': 'March 3, 2026', 'client_name': 'John Smith'})
tpl.save('/home/ubuntu/filled_template.docx')

# python-docx find-and-replace (for simple cases):
doc = Document('/path/to/template.docx')
find_and_replace(doc, '{{company_name}}', 'Acme Corp')
doc.save('/home/ubuntu/filled_template.docx')
```

## Embedding Matplotlib Charts

Generate a chart as PNG, then embed it directly from a BytesIO buffer — no temp file needed:

```python
import io
import matplotlib
import matplotlib.pyplot as plt
from docx import Document
from docx.shared import Inches

matplotlib.use('Agg')  # headless, no display required


def add_chart(doc, title: str, labels: list, values: list, chart_type: str = 'bar'):
    fig, ax = plt.subplots(figsize=(6, 3.5))
    if chart_type == 'bar':
        ax.bar(labels, values, color='#1a365d')
    elif chart_type == 'line':
        ax.plot(labels, values, marker='o', color='#1a365d')
    elif chart_type == 'pie':
        ax.pie(values, labels=labels, autopct='%1.1f%%')
    ax.set_title(title)
    ax.spines['top'].set_visible(False)
    ax.spines['right'].set_visible(False)
    plt.tight_layout()

    buf = io.BytesIO()
    fig.savefig(buf, format='png', dpi=150, bbox_inches='tight')
    plt.close(fig)
    buf.seek(0)

    doc.add_picture(buf, width=Inches(5.5))
    doc.paragraphs[-1].alignment = 1  # CENTER


# Usage:
doc = Document()
add_chart(doc, 'Q4 Revenue by Region',
          ['North', 'South', 'East', 'West'],
          [1.2, 0.8, 1.5, 0.9], chart_type='bar')
doc.save('/home/ubuntu/report.docx')
```
